[successioni di cauchy]

\begin{proposition} \label{pro:con-suc-mad}
    Sia $\left(\mathbb{X}, d\right)$ uno spazio metrico e $\left\{x_n\right\}_{n\in\mathbb{N}}$
    una successione di Cauchy in $\mathbb{X}$, allora
    \[
        \exists\left\{x_{n_k}\right\}_{k\in\mathbb{N}}\text{ sottosuccesione di }\left\{x_n\right\}
        \mid x_{n_k} \rightarrow p \implies x_n \rightarrow p.
    \]
\end{proposition}
