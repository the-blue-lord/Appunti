\begin{definition}\label{def:spazio-vettoriale}
    \textbf{Spazio vettoriale}

    Un qualsiasi insieme $\mathbb{X}\neq\emptyset$  è detto spazio vettoriale su $\mathbb{R}$
    se è dotato di un'operazione binaria di somma 
    \[ +:\mathbb{X}\times\mathbb{X} \rightarrow \mathbb{X} \]
    e di un'operazione di prodotto per uno scalare $\alpha\in\mathbb{R}$ 
    \[ \cdot :\mathbb{R}\times\mathbb{X} \rightarrow \mathbb{X} \]
    che godono delle seguenti proprietà caratteristiche,
    ovvero gli assiomi di uno spazio vettoriale.

    \begin{property}
        \textbf{Chiusura di $\mathbb{X}$ rispetto a somma e prodotto per scalare}
        \[ \forall\overline{x},\overline{y}\in\mathbb{X},
        \overline{x}+\overline{y}\in\mathbb{X}, \]
        \[ \forall\overline{x}\in\mathbb{X},\alpha\in\mathbb{R},
        \alpha\overline{x}\in\mathbb{X}, \]
    \end{property}
    \begin{property}
        \textbf{Commutatività della somma}
        \[ \forall\overline{x},\overline{y}\in\mathbb{X},
        \overline{x}+\overline{y}=\overline{y}+\overline{x}. \]
    \end{property}
    \begin{property}
        \textbf{Associatività della somma}
        \[ \forall\overline{x},\overline{y},\overline{z}\in\mathbb{X},
        \left(\overline{x}+\overline{y}\right)+\overline{z}=
        \overline{x}+\left(\overline{y}+\overline{z}\right). \]
    \end{property}
    \begin{property}
        \textbf{Esistenza dell'elemento neutro per la somma}
        \[ \exists\overline{0}\in\mathbb{X} \mid
        \forall \overline{x}\in\mathbb{X},  \overline{x}+\overline{0}
        =\overline{0}+\overline{x}=\overline{x}.\]
    \end{property}
    \begin{property}
        \textbf{Esistenza dell'inverso additivo}
        \[ \forall \overline{x}\in\mathbb{X},  \exists\overline{y}\in\mathbb{X} \mid
        \overline{x}+\overline{y}=\overline{y}+\overline{x}=\overline{0}.\]
    \end{property}

    \begin{property}
        \textbf{Associatività del prodotto per uno scalare}
        \[ \forall\overline{x}\in\mathbb{X},\, \alpha,\beta\in\mathbb{R},\;
        \alpha\left(\beta \overline{x}\right) = \left(\alpha\beta\right)\overline{x}. \]
    \end{property}
    \begin{property}
        \textbf{Esistenza dell'elemento neutro per il prodotto per uno scalare}
        \[ \forall \overline{x}\in\mathbb{X}, 1\overline{x}=\overline{x}.\]
    \end{property}
    \begin{property}
        \textbf{Distributività del prodotto per uno scalare sulla somma}
        \[ \forall\overline{x},\overline{y}\in\mathbb{X},\, \alpha\in\mathbb{R},\;
        \alpha\left(\overline{x}+\overline{y}\right) = \alpha\overline{x}+\alpha\overline{y}. \]
        
        \[ \forall\overline{x}\in\mathbb{X},\, \alpha,\beta\in\mathbb{R},\;
        \left(\alpha+\beta\right)\overline{x} = \alpha\overline{x}+\beta\overline{x}. \]
    \end{property}
\end{definition}
