\begin{definition}
    \textbf{Spazio vettoriale}

    Un qualsiasi insieme $\mathbb{X}$  è detto spazio vettoriale su $\mathbb{R}$
    se è chiuso rispetto ad un operazione di somma e ad una operazione di
    porodotto per uno scalare $\alpha\in\mathbb{R}$ che godono delle seguenti proprietà.

    \begin{property}
        \textbf{Commutatività della somma}
        \[ \forall\overline{x},\overline{y}\in\mathbb{X},
        \overline{x}+\overline{y}=\overline{x}+\overline{y}. \]
    \end{property}
    \begin{property}
        \textbf{Associatività della somma}
        \[ \forall\overline{x},\overline{y},\overline{z}\in\mathbb{X},
        \left(\overline{x}+\overline{y}\right)+\overline{z}=
        \overline{x}+\left(\overline{y}+\overline{z}\right). \]
    \end{property}
    \begin{property}
        \textbf{Esistenza dell'elemento neutro per la somma}
        \[ \forall \overline{x}\in\mathbb{X},  \exists\overline{p}=\left(0, 0,
        \ldots, 0\right)=\overline{0}\in\mathbb{X} \mid \overline{x}+\overline{p}
        =\overline{p}+\overline{x}=\overline{x}.\]
    \end{property}
    \begin{property}
        \textbf{Esistenza dell'elemento inverso per la somma}
        \[ \forall \overline{x}\in\mathbb{X},  \exists\overline{y}=\left(-x_1, -x_2,
        \ldots, -x_n\right)=-\overline{x}\in\mathbb{X} \mid \overline{x}+\overline{y}
        =\overline{y}+\overline{x}=\overline{0}=\overline{p}.\]
    \end{property}

    \begin{property}
        \textbf{Commutatività del prodotto per uno scalare}
        \[ \forall\overline{x}\in\mathbb{X}, \,\alpha\in\mathbb{R},\;
        \alpha \overline{x} = \overline{x} \alpha \]
    \end{property}
    \begin{property}
        \textbf{Associatività del prodotto per uno scalare}
        \[ \forall\overline{x}\in\mathbb{X},\, \alpha,\beta\in\mathbb{R},\;
        \alpha\left(\beta \overline{x}\right) = \left(\alpha\beta\right)\overline{x}. \]
    \end{property}
    \begin{property}
        \textbf{Distributività del prodotto per uno scalare sulla somma}
        \[ \forall\overline{x},\overline{y}\in\mathbb{X},\, \alpha\in\mathbb{R},\;
        \alpha\left(\overline{x}+\overline{y}\right) = \alpha\overline{x}+\alpha\overline{y}. \]
    \end{property}

\end{definition}
