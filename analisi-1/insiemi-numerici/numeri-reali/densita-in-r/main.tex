\begin{theorem} \label{teo:raz-den-rea}
    \textbf{Densità di $\mathbb{Q}$ in $\mathbb{R}$}

    $\mathbb{Q}$ è denso in $\mathbb{R}$, ovvero

    \[ \forall a, b \in \mathbb{R}, a < b \Rightarrow \exists r\in\mathbb{Q} \mid a<r<b. \]

    
\end{theorem}

\begin{corollary} \textit{di} \teoref{teo:raz-den-rea}

    Tra due numeri reali esistono infiniti numeri razionali.
\end{corollary}

\begin{theorem} \label{teo:irr-den-rea}
    \textbf{Densità di $\mathbb{R} \setminus\mathbb{Q}$ in $\mathbb{R}$}

    $\mathbb{R} \setminus\mathbb{Q}$ è denso in $\mathbb{R}$, ovvero

    \[ \forall a, b \in \mathbb{R}, a < b \Rightarrow \exists r\in\mathbb{R} \setminus\mathbb{Q} \mid a<r<b. \]
\end{theorem}
